const express = require('express') const app = express() const port = 3000 app.get('/', (req, res) => { res.send('Hello World') }) app.get('/dog', (req, res) => { res.json({ 'sound': '멍멍' }) }) app.get('/sound/:name', (req, res) => { const { name } = req.params if (name == "dog") { res.json({ 'sound': '멍멍' }) } }) app.listen(port, () => { console.log(`Example app listening on port ${port}`) }) //동물소리 api 서버 만들기 GET / sound/:name name따라서 다른 울움소리
